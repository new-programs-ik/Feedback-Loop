# Upper bounds matter here: this worker rebuilds from scratch on every deploy, so an unpinned
# dependency means a new major release can take production down without a single line of our code
# changing. That happened on 2026-08-27 — anthropic 1.x dropped `temperature` from Messages.create(),
# every analysis died with a TypeError. Bump these deliberately, with the tests green, not by accident.
anthropic>=0.115,<1.0     # 1.x is a breaking rewrite; 0.115.1 is the version the tests run against
fastapi>=0.110.0,<1.0
uvicorn>=0.29.0,<1.0
httpx>=0.27.0,<1.0
psycopg2-binary>=2.9,<3.0   # Postgres driver for store.py / ratings_store.py (the tests never open a DB)
pypdf>=4.0,<7.0             # class-materials extraction: slides/docs the analysis checks against
python-pptx>=0.6.23,<2.0
python-docx>=1.1,<2.0
imageio-ffmpeg>=0.5,<1.0    # static ffmpeg binary (with https) for the video-frames stage
google-auth>=2.29,<3.0      # service-account token for the ratings-sheet sync (token only; the
                            # Sheets call itself goes through httpx so tests can MockTransport it)
requests>=2.31,<3.0         # google-auth refreshes that token through its "requests transport", which
                            # imports this package. Missing here, the container built fine and every
                            # sync died with "The requests library is not installed" (14 + 16 Sep 2026).
                            # The tests never reach it (the token provider is faked), so only a real
                            # sync catches it.
reportlab>=4.2,<5.0         # the monthly / yearly PDF reports (reports.py)
matplotlib>=3.9,<4.0        # the charts in those reports (Agg backend, no display)
